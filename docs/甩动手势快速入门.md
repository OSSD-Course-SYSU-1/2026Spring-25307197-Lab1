# 甩动手势检测快速入门

## 一分钟快速使用

### 1. 导入组件

```typescript
import { FlingDetector, FlingCard, FlingDirection, FlingEvent } from '../commons/component_lib/Index';
```

### 2. 基础用法

```typescript
// 简单的甩动检测
FlingDetector({
  onFling: (event: FlingEvent) => {
    if (event.direction === FlingDirection.LEFT) {
      console.log('向左甩出');
    } else if (event.direction === FlingDirection.RIGHT) {
      console.log('向右甩出');
    }
  }
}) {
  Text('拖拽我')
    .width(200)
    .height(100)
    .backgroundColor('#4A90E2')
}
```

### 3. 卡片式UI（推荐）

```typescript
// Tinder风格的卡片
FlingCard({
  onFlingLeft: () => {
    console.log('拒绝');
  },
  onFlingRight: () => {
    console.log('接受');
  }
}) {
  Column() {
    Text('用户信息')
  }
}
.width(300)
.height(400)
```

## 常见场景

### 场景1: 列表项删除

```typescript
FlingCard({
  onFlingLeft: () => {
    // 删除项目
    this.deleteItem();
  }
}) {
  // 列表项内容
}
```

### 场景2: 分页导航

```typescript
FlingDetector({
  direction: PanDirection.Horizontal,
  onFling: (event) => {
    if (event.direction === FlingDirection.LEFT) {
      this.nextPage();
    } else if (event.direction === FlingDirection.RIGHT) {
      this.prevPage();
    }
  }
}) {
  // 页面内容
}
```

### 场景3: 四方向操作

```typescript
FlingCard({
  onFlingUp: () => { /* 归档 */ },
  onFlingDown: () => { /* 删除 */ },
  onFlingLeft: () => { /* 拒绝 */ },
  onFlingRight: () => { /* 接受 */ }
}) {
  // 卡片内容
}
```

## 参数调优

### 敏感模式（容易触发）

```typescript
FlingDetector({
  minFlingVelocity: 10,  // 降低速度阈值
  minFlingDistance: 50   // 降低距离阈值
})
```

### 精确模式（不易误触）

```typescript
FlingDetector({
  minFlingVelocity: 25,  // 提高速度阈值
  minFlingDistance: 150, // 提高距离阈值
  longPressDuration: 300 // 需要长按才能拖拽
})
```

## 核心概念

### FlingEvent 事件对象

```typescript
{
  direction: FlingDirection,  // 甩动方向
  velocity: number,           // 总速度
  velocityX: number,          // X方向速度
  velocityY: number,          // Y方向速度
  offsetX: number,            // X方向偏移
  offsetY: number,            // Y方向偏移
  duration: number            // 持续时间
}
```

### FlingDirection 方向枚举

```typescript
FlingDirection.UP    // 向上
FlingDirection.DOWN  // 向下
FlingDirection.LEFT  // 向左
FlingDirection.RIGHT // 向右
```
